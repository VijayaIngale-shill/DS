Step 5: Write the JavaScript code
Open script.js and paste this code:

const display = document.getElementById("display");

function appendValue(value) {
    display.value += value;
}

function clearDisplay() {
    display.value = "";
}

function deleteLast() {
    display.value = display.value.slice(0, -1);
}

function calculate() {
    try {
        if (!display.value.trim()) return;

        if (!/^[0-9+\-*/.() ]+$/.test(display.value)) {
            throw new Error("Invalid input");
        }

        const result = Function(
            '"use strict"; return (' + display.value + ')'
        )();

        if (!Number.isFinite(result)) {
            throw new Error("Invalid result");
        }

        display.value = result;

    } catch {
        display.value = "Error";
    }
}

If Live Server is not installed:
1.	Open the Extensions section in VS Code (Ctrl + Shift + X).
2.	Search for Live Server.
3.	Install the extension.
4.	Repeat the above steps.

Full Stack Practical 3: Conditional and List Rendering with Timeout

Step 1: Create a project folder
1. Open VS Code.
2. Create a new folder named Practical3.
3. Open the folder in VS Code.

Step 2: Create an HTML file
Inside the Practical3 folder, create one file:

index.html

Your folder structure should look like this:

Practical3
--index.html

Step 3: Write the HTML and AngularJS code
Open index.html and paste this complete code:

<!DOCTYPE html>
<html ng-app="myApp">

<head>
    <title>Conditional and List Rendering</title>

    <script src="https://ajax.googleapis.com/ajax/libs/angularjs/1.8.2/angular.min.js"></script>
</head>

<body ng-controller="myCtrl">

    <h2>Conditional Rendering</h2>

    <button ng-click="show = !show">
        Show / Hide
    </button>

    <p ng-if="show">Welcome to Full Stack!</p>


    <h2>List Rendering</h2>

    <ul>
        <li ng-repeat="item in items">
            {{item}}
        </li>
    </ul>

    <h3>{{message}}</h3>


    <script>

        var app = angular.module("myApp", []);

        app.controller("myCtrl", function($scope, $timeout) {

            $scope.show = true;

            $scope.items = [
                "HTML",
                "CSS",
                "JavaScript",
                "AngularJS"
            ];

            $scope.message = "Please wait...";

            $timeout(function() {
                $scope.message = "Timeout completed!";
            }, 3000);

        });

    </script>

</body>
</html>

Full Stack Practical 4: Routing, Stopwatch and Timeout

Step 1: Create a project folder
1.	Open VS Code.
2.	Create a new folder named Practical4.
3.	Open the folder in VS Code.

Step 2: Create an HTML file
Create one file inside the folder:
index.html
Your folder structure:
Practical4
index.html

Step 3: Write the complete code
Open index.html and paste this code:
<!DOCTYPE html>
<html ng-app="myApp">

<head>
    <title>Routing Stopwatch</title>

    <script src="https://ajax.googleapis.com/ajax/libs/angularjs/1.8.2/angular.min.js"></script>

    <script src="https://ajax.googleapis.com/ajax/libs/angularjs/1.8.2/angular-route.min.js"></script>

</head>

<body ng-controller="myCtrl">

    <h2>AngularJS Practical 4</h2>

    <button ng-click="page='Home'">Home</button>

    <button ng-click="page='About'">About</button>

    <h3>{{page}}</h3>

    <h3>Stopwatch: {{seconds}} seconds</h3>

    <button ng-click="start()">Start</button>

    <button ng-click="stop()">Stop</button>

    <button ng-click="reset()">Reset</button>

    <p>{{message}}</p>

    <script>

        var app = angular.module('myApp', []);

        app.controller('myCtrl', function($scope, $interval, $timeout) {

            $scope.page = 'Home';

            $scope.seconds = 0;

            $scope.message = '';

            var timer;

            $scope.start = function() {

                if (!timer) {

                    timer = $interval(function() {

                        $scope.seconds++;

                    }, 1000);

                }

            };

            $scope.stop = function() {

                if (timer) {

                    $interval.cancel(timer);

                    timer = null;

                }

            };

            $scope.reset = function() {

                $scope.stop();

                $scope.seconds = 0;

            };

            $timeout(function() {

                $scope.message = 'Welcome to AngularJS!';

            }, 2000);

        });

    </script>

</body>
</html>
Save the file using Ctrl + S.

Step 5: Run the practical
1.	Right-click on index.html.
2.	Select Open with Live Server.
3.	Your browser will open automatically.
Alternatively, double-click index.html from File Explorer.
Internet connection is required to load the AngularJS libraries from Google's CDN.

Step 6: Test the output

from flask import Flask, request, jsonify
import datetime

app = Flask(__name__)

todos = []
next_id = 1


# CREATE TODO - POST
@app.route('/todos', methods=['POST'])
def create_todo():
    global next_id

    data = request.get_json()
    task = data.get('task')
    status = data.get('status')

    if task is None:
        return jsonify({'error': 'Task is required'}), 400

    date = datetime.datetime.now().strftime("%Y-%m-%d %H:%M:%S")

    new_todo = {
        'id': next_id,
        'task': task,
        'date': date,
        'status': status
    }

    todos.append(new_todo)
    next_id += 1

    return "Task added", 201


# GET ALL TODOS
@app.route('/todos', methods=['GET'])
def get_todos():
    return jsonify(todos)


# GET TODO BY ID
@app.route('/todos/<int:todo_id>', methods=['GET'])
def get_todo(todo_id):

    todo = next(
        (todo for todo in todos if todo['id'] == todo_id),
        None
    )

    if todo is None:
        return jsonify({'error': 'Todo not found'}), 404

    return jsonify(todo)


# UPDATE TODO - PUT
@app.route('/todos/<int:todo_id>', methods=['PUT'])
def update_todo(todo_id):

    data = request.get_json()
    task = data.get('task')

    if task is None:
        return jsonify({'error': 'Task is required'}), 400

    todo = next(
        (todo for todo in todos if todo['id'] == todo_id),
        None
    )

    if todo is None:
        return jsonify({'error': 'Todo not found'}), 404

    todo['task'] = task

    return jsonify(todo)


# DELETE TODO
@app.route('/todos/<int:todo_id>', methods=['DELETE'])
def delete_todo(todo_id):
    global todos

    todo = next(
        (todo for todo in todos if todo['id'] == todo_id),
        None
    )

    if todo is None:
        return jsonify({'error': 'Todo not found'}), 404

    todos = [
        todo for todo in todos
        if todo['id'] != todo_id
    ]

    return jsonify({"message": "Task Deleted"})


if __name__ == "__main__":
    app.run(debug=True)

const express = require("express");
const axios = require("axios");

const app = express();
const port = 3000;

app.use(express.json());

let healthdatas = [];

// GET API - View all health records
app.get("/healthtracker", (req, res) => {
    res.send(healthdatas);
});

// POST API - Create health tracker
app.post("/healthcare", (req, res) => {

    const { title, duration } = req.body;

    if (!title || !duration) {
        return res.status(400).json({
            message: "Please provide title and duration"
        });
    }

    const healthdata = {
        id: healthdatas.length + 1,
        title,
        duration
    };

    // Send data to Flask To-Do API
    axios.post("http://127.0.0.1:5000/todos", {
        task: title,
        status: "completed"
    })
    .then((response) => {
        console.log("Response from Flask:", response.data);
    })
    .catch((error) => {
        console.log(error.message);
    });

    healthdatas.push(healthdata);

    res.status(201).json(healthdata);
});

// Start Express server
app.listen(port, () => {
    console.log("Server running at http://localhost:3000");
});

import pika

# Connect to RabbitMQ
connection = pika.BlockingConnection(
    pika.ConnectionParameters('localhost')
)

channel = connection.channel()

# Create a queue
channel.queue_declare(queue='hello', durable=True)

# Message to send
message = "Hello, RabbitMQ!"

# Send message to queue
channel.basic_publish(
    exchange='',
    routing_key='hello',
    body=message
)

print(f"[x] Sent '{message}'")

# Close connection
connection.close()


step 5:- Open Receive.py and paste:


import pika

# Connect to RabbitMQ
connection = pika.BlockingConnection(
    pika.ConnectionParameters('localhost')
)

channel = connection.channel()

# Declare the same queue
channel.queue_declare(queue='hello', durable=True)

# Function to receive message
def callback(ch, method, properties, body):
    print(f"[x] Received {body.decode()}")

    # Acknowledge message
    ch.basic_ack(delivery_tag=method.delivery_tag)

# Receive messages
channel.basic_consume(
    queue='hello',
    on_message_callback=callback
)

print('[*] Waiting for messages. To exit press CTRL+C')

# Start receiving
channel.start_consuming()

4.	Restart your computer if required.

Step 3: Check installation
Open CMD and execute:
>> vagrant --version
You should see the installed Vagrant version.


Step 4: Create project folder
>> mkdir vagrant_vm
>> cd vagrant_vm


Step 5: Initialize Vagrant
Execute:
>> vagrant init

This creates a file named Vagrantfile.

Step 6: Configure Vagrantfile
Open Vagrantfile in VS Code.

Replace its contents with:

Vagrant.configure("2") do |config|

  config.vm.box = "ubuntu/bionic64"

  config.vm.network "private_network",
    ip: "192.168.56.10"

  config.vm.provider "virtualbox" do |vb|
    vb.memory = 1024
    vb.cpus = 2
  end

end

Save the file.

This configuration specifies:
•	Ubuntu as the operating system.
•	Private IP address: 192.168.56.10.
•	RAM: 1024 MB (1 GB).
•	CPU: 2 cores.


Step 7: Start the Virtual Machine
Execute:
>> vagrant up

Vagrant will download the Ubuntu image if necessary and create the VM in VirtualBox.
VM started

Step 8: Connect to the Virtual Machine
Execute:
>> vagrant ssh

You will enter the Ubuntu terminal.

Step 9: Exit the Virtual Machine
Execute:
>>exit

This disconnects your terminal from the Ubuntu VM.
